SQL — crear tabla (una sola vez)

Corré esto en el SQL Editor de Supabase:

-- Corré esto si es la primera vez:
create table if not exists rendiciones (
  _id bigint primary key,
  fecha text, cajera text,
  efectivo_total numeric default 0,
  visa_credito numeric default 0,
  visa_debito numeric default 0,
  amex numeric default 0,
  mastercard numeric default 0,
  maestro numeric default 0,
  naranja numeric default 0,
  transferencias numeric default 0,
  or_patagway numeric default 0,
  cuenta_dni numeric default 0,
  total_general numeric default 0,
  _guardado text
);
alter table rendiciones enable row level security;
create policy "acceso anonimo" on rendiciones
  for all using (true) with check (true);

-- Si ya tenías la tabla vieja, corré esto en su lugar:
-- drop table rendiciones;
-- y volvé a crear con el CREATE de arriba.
