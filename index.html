SQL — crear tabla (una sola vez)

Corré esto en el SQL Editor de Supabase:

create table rendiciones (
  _id bigint primary key,
  fecha text, cajera text, local text, turno text,
  pit numeric default 0, cierre_z numeric default 0,
  nota_credito numeric default 0,
  venta_accesorios numeric default 0,
  venta_celulares numeric default 0,
  venta_total numeric default 0,
  efectivo numeric default 0, tarjetas numeric default 0,
  transferencias numeric default 0,
  or_patagway numeric default 0,
  cuenta_dni numeric default 0,
  visa_credito numeric default 0,
  mastercard numeric default 0, maestro numeric default 0,
  naranja numeric default 0, visa_debito numeric default 0,
  amex numeric default 0, total_gastos numeric default 0,
  subo_comision numeric default 0,
  total_rendido numeric default 0,
  diferencia numeric default 0,
  observaciones text, _guardado text
);
alter table rendiciones enable row level security;
create policy "acceso anonimo" on rendiciones
  for all using (true) with check (true);
